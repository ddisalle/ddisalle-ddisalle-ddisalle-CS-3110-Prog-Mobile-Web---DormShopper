<script>

let currentUser = null;


// ============================================================
// LOAD USER
// ============================================================

async function loadUser() {

    try {

        const response = await fetch("/api/me");

        const data = await response.json();

        if (data.loggedIn) {
            currentUser = data.user;
        }

    } catch (error) {

        console.error(
            "Unable to check login status:",
            error
        );

    }

}


// ============================================================
// LOAD LISTINGS
// ============================================================

async function loadListings() {

    const container =
        document.getElementById("listingContainer");

    try {

        const response =
            await fetch("/api/listings");

        if (!response.ok) {
            throw new Error(
                "Unable to load listings."
            );
        }

        const listings =
            await response.json();

        container.innerHTML = "";


        if (listings.length === 0) {

            container.innerHTML = `
                <div class="empty-state">
                    <h2>No Items Yet</h2>
                    <p>
                        Be the first person to post an item!
                    </p>
                    <a href="/post" class="button">
                        Post an Item
                    </a>
                </div>
            `;

            return;
        }


        listings.forEach(listing => {

            const card =
                document.createElement("div");

            card.className =
                "listing-card";


            let imageHTML = "";

            if (listing.image) {

                imageHTML = `
                    <img
                        src="${escapeHTML(listing.image)}"
                        alt="${escapeHTML(listing.title)}"
                        class="listing-image"
                        onerror="this.style.display='none'"
                    >
                `;

            }


            let buttons = "";

            if (
                currentUser &&
                currentUser.id === listing.user_id
            ) {

                buttons = `
                    <div class="listing-actions">

                        <button
                            class="edit-button"
                            onclick="editListing(${listing.id})"
                        >
                            Edit
                        </button>

                        <button
                            class="delete-button"
                            onclick="deleteListing(${listing.id})"
                        >
                            Delete
                        </button>

                    </div>
                `;

            }


            card.innerHTML = `

                ${imageHTML}

                <div class="listing-content">

                    <h2>
                        ${escapeHTML(listing.title)}
                    </h2>

                    <p class="price">
                        $${Number(listing.price).toFixed(2)}
                    </p>

                    ${
                        listing.category
                        ?
                        `<p class="category">
                            ${escapeHTML(listing.category)}
                        </p>`
                        :
                        ""
                    }

                    <p>
                        ${escapeHTML(
                            listing.description || ""
                        )}
                    </p>

                    <p class="seller">
                        Posted by:
                        ${escapeHTML(
                            listing.username
                        )}
                    </p>

                    ${buttons}

                </div>

            `;


            container.appendChild(card);

        });


    } catch (error) {

        console.error(error);

        container.innerHTML = `
            <div class="error-message">
                <h2>Unable to load listings</h2>
                <p>
                    Please refresh the page and try again.
                </p>
            </div>
        `;

    }

}


// ============================================================
// EDIT LISTING
// ============================================================

async function editListing(id) {

    try {

        const response =
            await fetch(`/api/listings/${id}`);

        const listing =
            await response.json();


        if (!response.ok) {

            alert(
                listing.message ||
                "Unable to load listing."
            );

            return;
        }


        document.getElementById(
            "editId"
        ).value = listing.id;


        document.getElementById(
            "editTitle"
        ).value = listing.title;


        document.getElementById(
            "editDescription"
        ).value = listing.description || "";


        document.getElementById(
            "editPrice"
        ).value = listing.price;


        document.getElementById(
            "editCategory"
        ).value = listing.category || "";


        document.getElementById(
            "editImage"
        ).value = listing.image || "";


        document.getElementById(
            "editModal"
        ).style.display = "block";


    } catch (error) {

        console.error(error);

        alert(
            "Unable to load listing."
        );

    }

}


// ============================================================
// CLOSE EDIT MODAL
// ============================================================

function closeEditModal() {

    document.getElementById(
        "editModal"
    ).style.display = "none";

}


// ============================================================
// SAVE EDIT
// ============================================================

document
    .getElementById("editForm")
    .addEventListener("submit", async function(event) {

        event.preventDefault();


        const id =
            document.getElementById(
                "editId"
            ).value;


        const listing = {

            title:
                document.getElementById(
                    "editTitle"
                ).value,

            description:
                document.getElementById(
                    "editDescription"
                ).value,

            price:
                document.getElementById(
                    "editPrice"
                ).value,

            category:
                document.getElementById(
                    "editCategory"
                ).value,

            image:
                document.getElementById(
                    "editImage"
                ).value

        };


        try {

            const response =
                await fetch(
                    `/api/listings/${id}`,
                    {
                        method: "PUT",

                        headers: {
                            "Content-Type":
                                "application/json"
                        },

                        body:
                            JSON.stringify(listing)
                    }
                );


            const data =
                await response.json();


            if (!response.ok) {

                alert(
                    data.message ||
                    "Unable to update listing."
                );

                return;
            }


            alert(
                "Listing updated successfully!"
            );


            closeEditModal();

            await loadListings();


        } catch (error) {

            console.error(error);

            alert(
                "Unable to update listing."
            );

        }

    });


// ============================================================
// DELETE LISTING
// ============================================================

async function deleteListing(id) {

    const confirmed =
        confirm(
            "Are you sure you want to delete this listing?"
        );


    if (!confirmed) {
        return;
    }


    try {

        const response =
            await fetch(
                `/api/listings/${id}`,
                {
                    method: "DELETE"
                }
            );


        const data =
            await response.json();


        if (!response.ok) {

            alert(
                data.message ||
                "Unable to delete listing."
            );

            return;
        }


        alert(
            "Listing deleted successfully."
        );


        await loadListings();


    } catch (error) {

        console.error(error);

        alert(
            "Unable to delete listing."
        );

    }

}


// ============================================================
// ESCAPE HTML
// ============================================================

function escapeHTML(value) {

    const div =
        document.createElement("div");

    div.textContent =
        value ?? "";

    return div.innerHTML;

}


// ============================================================
// CLOSE MODAL WHEN CLICKING OUTSIDE
// ============================================================

window.onclick = function(event) {

    const modal =
        document.getElementById("editModal");

    if (event.target === modal) {
        closeEditModal();
    }

};

})();
