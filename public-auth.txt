document.addEventListener("DOMContentLoaded", async function () {

    const authLinks =
        document.getElementById("authLinks");


    if (!authLinks) {
        return;
    }


    try {

        const response =
            await fetch("/api/me");


        const data =
            await response.json();


        if (data.loggedIn) {

            authLinks.innerHTML = `

                <span class="welcome">
                    Hi, ${escapeAuthHTML(data.user.username)}
                </span>

                <button
                    class="logout-button"
                    onclick="logoutUser()"
                >
                    Logout
                </button>

            `;

        } else {

            authLinks.innerHTML = `

                <a href="/login">
                    Login
                </a>

            `;

        }


    } catch (error) {

        console.error(
            "Authentication check failed:",
            error
        );

    }

});


async function logoutUser() {

    try {

        const response =
            await fetch(
                "/api/logout",
                {
                    method: "POST"
                }
            );


        if (!response.ok) {

            alert(
                "Unable to log out."
            );

            return;
        }


        window.location.href =
            "/";


    } catch (error) {

        console.error(error);

        alert(
            "Unable to log out."
        );

    }

}


function escapeAuthHTML(value) {

    const div =
        document.createElement("div");

    div.textContent =
        value ?? "";

    return div.innerHTML;

}
