<script>

document
    .getElementById("postForm")
    .addEventListener("submit", async function(event) {

        event.preventDefault();


        const message =
            document.getElementById("message");


        const listing = {

            title:
                document.getElementById(
                    "title"
                ).value,

            description:
                document.getElementById(
                    "description"
                ).value,

            price:
                document.getElementById(
                    "price"
                ).value,

            category:
                document.getElementById(
                    "category"
                ).value,

            image:
                document.getElementById(
                    "image"
                ).value

        };


        try {

            const response =
                await fetch(
                    "/api/listings",
                    {
                        method: "POST",

                        headers: {
                            "Content-Type":
                                "application/json"
                        },

                        body:
                            JSON.stringify(listing)
                    }
                );


            const data =
                await response.json();


            if (!response.ok) {

                message.textContent =
                    data.message ||
                    "Unable to post item.";

                message.className =
                    "form-message error";

                return;
            }


            message.textContent =
                "Item posted successfully!";


            message.className =
                "form-message success";


            document
                .getElementById("postForm")
                .reset();


            setTimeout(() => {

                window.location.href =
                    "/browse";

            }, 1000);


        } catch (error) {

            console.error(error);

            message.textContent =
                "Unable to connect to the server.";

            message.className =
                "form-message error";

        }

    });

</script>
